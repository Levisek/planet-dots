# V4.4 F1 — Poziční jádro (Implementation Plan)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Přepnout zdroj pozic všech těles na jednotný `positionProvider` (astronomy-engine pro krytá tělesa, Kepler z reálných elementů pro zbytek) a přepsat mód na pouhou zobrazovací transformaci — bez viditelné změny narativu, ale s poziční pravdou a orbitálními čarami, které konečně sedí.

**Architecture:** Reálná heliocentrická pozice (ekliptika J2000, AU) → převod do scene frame → mode transform (Fyzikální lineární / Pochopení reálný směr + komprimovaný poloměr) → anchor.position. Dva pozicové backendy za jedním interface.

**Tech Stack:** Vanilla JS ES modules, Three.js 0.184 (vendored), astronomy-engine (vendored, VSOP87), `node --test`.

## Global Constraints

- **Offline soběstačnost:** astronomy-engine se VENDORUJE do `vendor/astronomy/`, žádný CDN. CSP `default-src 'none'` + `script-src 'self'` už nový vendored ESM soubor pokrývá (je to `self`).
- **Frame konvence:** Provider vrací **scene frame** (Y nahoru = ekliptický sever, orbitální rovina X-Z), v **AU**. `coordinateFrame.eclToScene` se používá jen uvnitř ephemeris backendu.
- **Zpětná kompatibilita orbit.js:** `orbitPosition(t, phase, period, a, e, incDeg=0)` NESMÍ změnit chování pro existující volání — nové parametry `OmegaDeg=0, omegaDeg=0` se přidají NA KONEC jako optional.
- **Žádné vymyšlené pole:** `e` v datech = reálná hodnota (dnešní `eReal`), staré vymyšlené `e` a `period`/`periodReal` se mažou. Sinope/Pasiphae `a` (commit ded2171) se revertuje na V4.3 hodnoty (16.0 / 15.5).
- **Commit message česky, bez footerů**, jen pod jménem uživatele.
- **121 testů** nesmí regresovat kromě těch, co testují mazané clampy (Task 9 je přepíše).

```js
/** Rotace kolem osy Y o `deg` stupňů. */
export function rotateY(pos, deg) {
  if (deg === 0) return { x: pos.x, y: pos.y, z: pos.z };
  const r = (deg * Math.PI) / 180;
  const c = Math.cos(r), s = Math.sin(r);
  return { x: pos.x * c + pos.z * s, y: pos.y, z: -pos.x * s + pos.z * c };
}
```

A přepsat `orbitPosition` (zachovat existující tělo, přidat rotace):

```js
export function orbitPosition(t, phaseOffset, period, a, e, incDeg = 0, OmegaDeg = 0, omegaDeg = 0) {
  const M = (2 * Math.PI * t) / period + phaseOffset;
  const E = solveKepler(M, e);
  const xp = a * (Math.cos(E) - e);
  const zp = a * Math.sqrt(1 - e * e) * Math.sin(E);
  // perifocal (periapsis +X, rovina X-Z) → ω (kolem Y) → inc (kolem X) → Ω (kolem Y)
  let p = rotateY({ x: xp, y: 0, z: zp }, omegaDeg);
  p = applyInclination(p, incDeg);
  p = rotateY(p, OmegaDeg);
  return { x: p.x, y: p.y, z: p.z, E };
}
```

```js
import { orbitPosition } from './orbit.js';

const J2000_MS = Date.UTC(2000, 0, 1, 12, 0, 0);
const MS_PER_DAY = 86400000;

// Elementy referencované k ekliptice J2000; a v AU. Vrací scene-frame AU
// (orbit.js pracuje v Y-up X-Z = scene frame). M0 v epoše J2000.
export function keplerHelio(el, date) {
  const days = (date.getTime() - J2000_MS) / MS_PER_DAY;
  const period = el.periodDays;
  const phase = (el.M0Deg * Math.PI) / 180;
  const { x, y, z } = orbitPosition(
    days, phase, period, el.aAU, el.e, el.incDeg, el.OmegaDeg, el.omegaDeg,
  );
  return { x, y, z };
}
```

```js
import * as Astronomy from '../vendor/astronomy/astronomy.js';
import { eclToScene } from './coordinateFrame.js';

const PLANET_BODY = {
  sun: Astronomy.Body.Sun, mercury: Astronomy.Body.Mercury,
  venus: Astronomy.Body.Venus, earth: Astronomy.Body.Earth,
  mars: Astronomy.Body.Mars, jupiter: Astronomy.Body.Jupiter,
  saturn: Astronomy.Body.Saturn, uranus: Astronomy.Body.Uranus,
  neptune: Astronomy.Body.Neptune,
};

// EQJ vektor → ekliptika J2000 → scene frame.
function eqjToScene(v) {
  const ecl = Astronomy.Ecliptic(v); // {vec:{x,y,z}, elat, elon} v ekliptice J2000
  return eclToScene({ x: ecl.vec.x, y: ecl.vec.y, z: ecl.vec.z });
}

export function ephemHelio(bodyId, date) {
  const body = PLANET_BODY[bodyId];
  if (body === Astronomy.Body.Sun) return { x: 0, y: 0, z: 0 };
  const t = Astronomy.MakeTime(date);
  return eqjToScene(Astronomy.HelioVector(body, t));
}

export function ephemMoonRelative(moonId, date) {
  const t = Astronomy.MakeTime(date);
  if (moonId === 'luna') {
    const g = Astronomy.GeoVector(Astronomy.Body.Moon, t, false); // geocentr. EQJ, AU
    return eqjToScene(g);
  }
  const jm = Astronomy.JupiterMoons(t); // stavové vektory vůči Jupiteru, EQJ, AU
  const s = jm[moonId];
  return eqjToScene({ x: s.x, y: s.y, z: s.z });
}
```

```js
// Explicitní krytí: která tělesa jedou z ephemeris (astronomy-engine) a která

export const EPHEMERIS_BODIES = new Set([
  'sun', 'mercury', 'venus', 'earth', 'mars', 'jupiter', 'saturn', 'uranus', 'neptune',
]);
export const EPHEMERIS_MOONS = new Set(['luna', 'io', 'europa', 'ganymede', 'callisto']);

export function sourceOf(bodyId) {
  return (EPHEMERIS_BODIES.has(bodyId) || EPHEMERIS_MOONS.has(bodyId))

}
```

```js
import { auToDisplayRadius } from './planetOrbits.js';

const LINEAR_AU = 3846;

export function toDisplay(pAU) {
  if (isFyzikalni()) {
    return { x: pAU.x * LINEAR_AU, y: pAU.y * LINEAR_AU, z: pAU.z * LINEAR_AU };
  }
  const r = Math.hypot(pAU.x, pAU.y, pAU.z);
  if (r === 0) return { x: 0, y: 0, z: 0 };
  const rDisp = auToDisplayRadius(r);
  const k = rDisp / r;
  return { x: pAU.x * k, y: pAU.y * k, z: pAU.z * k };
}
```

```js
const MOON_LINEAR_AU = 3846;      // Fyzikální stejné měřítko
const MOON_COMPRESS = 60000;      // Pochopení: AU → px vůči rodiči (kalibrovat v F1)
export function toDisplayRelative(relAU) {
  if (isFyzikalni()) {
    return { x: relAU.x * MOON_LINEAR_AU, y: relAU.y * MOON_LINEAR_AU, z: relAU.z * MOON_LINEAR_AU };
  }
  const r = Math.hypot(relAU.x, relAU.y, relAU.z);
  if (r === 0) return { x: 0, y: 0, z: 0 };
  const rDisp = Math.sqrt(r) * MOON_COMPRESS; // kompaktní, monotónní
  const k = rDisp / r;
  return { x: relAU.x * k, y: relAU.y * k, z: relAU.z * k };
}
```
